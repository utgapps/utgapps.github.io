37// Turn a status number into a sentence a person can act on.
38function explain(status, problem) {
39 if (status === 401) { return 'The server does not recognise your key. Check it for typos.'; }
40 if (status === 429) { return 'You are sending too fast. Wait a few seconds and try again.'; }
41 if (status === 400) { return 'The server could not use that request: ' + problem.error.message; }
42 return 'Something went wrong on the server (' + status + '). Try again in a moment.';
43}

55// Week 8: the server reads at most 4000 characters of conversation at once.
56// Past that it refuses the whole request, so old messages have to go.
57const MAX_CHARS = 3500; // leave room for the personality and your next line
58 
59function trimHistory() {
60 let size = personaBox.value.length;
61 for (const message of history) {
62 size += message.content.length;
63 }
64 // Drop from the FRONT. The oldest messages matter least - usually.
65 while (size > MAX_CHARS && history.length > 2) {
66 const dropped = history.shift();
67 size -= dropped.content.length;
68 console.log('Forgot an old message to stay under the limit.');
69 }
70}

72// Week 10: with stream:true the answer arrives in pieces while it is still
73// being written. Same words, same total time - they just start sooner.
74//
75// Week 11: onPiece is a FUNCTION we are handed. Every time a bit of text
76// turns up we call it, and whoever gave it to us decides what to do.
77async function readStream(res, onPiece) {
78 const reader = res.body.getReader();
79 const decoder = new TextDecoder();
80 let buffer = ''; // a piece can be cut in half mid-arrival
81 let answer = '';
82 
83 while (true) {
84 const chunk = await reader.read();
85 if (chunk.done) { break; }
86 buffer += decoder.decode(chunk.value, { stream: true });
87 
88 // Complete pieces are separated by a blank line. Keep the last,
89 // possibly unfinished, piece in the buffer for the next round.
90 const parts = buffer.split('\n\n');
91 buffer = parts.pop();
92 
93 for (const part of parts) {
94 const line = part.trim();
95 if (!line.startsWith('data: ')) { continue; }
96 const payload = line.slice(6);
97 if (payload === '[DONE]') { continue; }
98 const piece = JSON.parse(payload);
99 const bit = piece.choices[0].delta.content;
100 if (bit) {
101 answer += bit;
102 onPiece(answer); // week 11: show it NOW, do not wait for the end
103 }
104 }
105 }
106 return answer;
107}

109// Week 4: ask the AI a real question.
110// A chat request is a LIST of messages. Each one says who said it.
111async function askAI(onPiece) {
112 trimHistory(); // week 8: never send more than the server will read
113 const res = await fetch(AI_BASE + '/v1/chat/completions', {
114 method: 'POST', // POST = I am sending you something
115 headers: {
116 'Content-Type': 'application/json', // what I am sending is JSON
117 'Authorization': 'Bearer ' + API_KEY
118 },
119 body: JSON.stringify({
120 model: modelBox.value, // week 9: chosen on the page
121 temperature: Number(tempBox.value), // 0 = careful, 1.5 = wild
122 stream: true, // week 10: send it in pieces
123 messages: [
124 // Week 6: the standing instruction, read before every single reply.
125 { role: 'system', content: personaBox.value },
126 // Week 7: everything said so far, oldest first.
127 ...history
128 ]
129 })
130 });
131 
132 // Week 5: fetch does NOT throw when the server says no. A 429 arrives
133 // looking just like a success until you actually check.
134 if (!res.ok) {
135 const problem = await res.json();
136 throw new Error(explain(res.status, problem));
137 }
138 
139 return readStream(res, onPiece); // hand the drawing job through
140}
