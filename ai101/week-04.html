32// Week 4: ask the AI a real question.
33// A chat request is a LIST of messages. Each one says who said it.
34async function askAI(question) {
35 const res = await fetch(AI_BASE + '/v1/chat/completions', {
36 method: 'POST', // POST = I am sending you something
37 headers: {
38 'Content-Type': 'application/json', // what I am sending is JSON
39 'Authorization': 'Bearer ' + API_KEY
40 },
41 body: JSON.stringify({
42 model: 'fast',
43 messages: [
44 { role: 'user', content: question }
45 ]
46 })
47 });
48 const data = await res.json();
49 return data.choices[0].message.content;
50}

62 addLine('Companion', reply);
63});
