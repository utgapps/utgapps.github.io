55// Turn a status number into a sentence a person can act on.
56function explain(status, problem) {
57 // The server usually explains itself better than we can guess. Use its
58 // words when it gives us any, and fall back to our own when it does not.
59 const said = problem && problem.error && problem.error.message;
60 if (said) { return said; }
61 if (status === 401) { return 'The server does not recognise your key. Check it for typos.'; }
62 if (status === 429) { return 'You are sending too fast. Wait a few seconds and try again.'; }
63 return 'Something went wrong on the server (' + status + '). Try again in a moment.';
64}

80function trimHistory() {
81 let size = personaBox.value.length;
82 for (const message of history) {
83 size += message.content.length;
84 }
85 // Drop from the FRONT. The oldest messages matter least - usually.
86 while (size > MAX_CHARS && history.length > 2) {
87 const dropped = history.shift();
88 size -= dropped.content.length;
89 console.log('Forgot an old message to stay under the limit.');
90 }

93// Week 10: with stream:true the answer arrives in pieces while it is still
94// being written. Same words, same total time - they just start sooner.
95//
96// Week 11: onPiece is a FUNCTION we are handed. Every time a bit of text
97// turns up we call it, and whoever gave it to us decides what to do.
98async function readStream(res, onPiece) {
99 const reader = res.body.getReader();
100 const decoder = new TextDecoder();
101 let buffer = ''; // a piece can be cut in half mid-arrival
102 let answer = '';
103 
104 while (true) {
105 const chunk = await reader.read();
106 if (chunk.done) { break; }
107 buffer += decoder.decode(chunk.value, { stream: true });
108 
109 // Complete pieces are separated by a blank line. Keep the last,
110 // possibly unfinished, piece in the buffer for the next round.
111 const parts = buffer.split('\n\n');
112 buffer = parts.pop();
113 
114 for (const part of parts) {
115 const line = part.trim();
116 if (!line.startsWith('data: ')) { continue; }
117 const payload = line.slice(6);
118 if (payload === '[DONE]') { continue; }
119 const piece = JSON.parse(payload);
120 const bit = piece.choices[0].delta.content;
121 if (bit) {
122 answer += bit;
123 onPiece(answer); // week 11: show it NOW, do not wait for the end
124 }
125 }
126 }
127 return answer;
128}

130// Week 4: ask the AI a real question.
131// A chat request is a LIST of messages. Each one says who said it.
132async function askAI(onPiece) {
133 trimHistory(); // week 8: never send more than the server will read
134 const res = await fetch(AI_BASE + '/v1/chat/completions', {
135 method: 'POST', // POST = I am sending you something
136 headers: {
137 'Content-Type': 'application/json', // what I am sending is JSON
138 'Authorization': 'Bearer ' + API_KEY
139 },
140 body: JSON.stringify({
141 model: modelBox.value, // week 9: chosen on the page
142 temperature: Number(tempBox.value), // 0 = careful, 1.5 = wild
143 stream: true, // week 10: send it in pieces
144 messages: [
145 // Week 6: the standing instruction, read before every single reply.
146 { role: 'system', content: personaBox.value },
147 // Week 7: everything said so far, oldest first.
148 ...history
149 ]
150 })
151 });
152 
153 // Week 5: fetch does NOT throw when the server says no. A 429 arrives
154 // looking just like a success until you actually check.
155 if (!res.ok) {
156 const problem = await res.json();
157 throw new Error(explain(res.status, problem));
158 }
159 
160 return readStream(res, onPiece); // hand the drawing job through
161}
