23// Put one message into the chat box as a bubble.
24// 'who' is 'you', 'bot' or 'problem' - it picks the style.
25function addLine(who, text) {
26 const bubble = document.createElement('div');
27 bubble.className = 'bubble ' + who;
28 bubble.textContent = text;
29 chat.appendChild(bubble);
30 chat.scrollTop = chat.scrollHeight; // keep the newest message in view
31 return bubble;
32}

34// Turn a status number into a sentence a person can act on.
35function explain(status, problem) {
36 if (status === 401) { return 'The server does not recognise your key. Check it for typos.'; }
37 if (status === 429) { return 'You are sending too fast. Wait a few seconds and try again.'; }
38 if (status === 400) { return 'The server could not use that request: ' + problem.error.message; }
39 return 'Something went wrong on the server (' + status + '). Try again in a moment.';
40}

42// The smallest useful request: what models does this server have?
43// 'async' means this function does something slow and can be waited for.
44async function listModels() {
45 const res = await fetch(AI_BASE + '/v1/models', {
46 headers: { 'Authorization': 'Bearer ' + API_KEY }
47 });
48 const data = await res.json(); // the answer arrives as text; this reads it
49 console.log('The server offers:', data.data);
50}

52// Week 8: the server reads at most 4000 characters of conversation at once.
53// Past that it refuses the whole request, so old messages have to go.
54const MAX_CHARS = 3500; // leave room for the personality and your next line
55 
56function trimHistory() {
57 let size = personaBox.value.length;
58 for (const message of history) {
59 size += message.content.length;
60 }
61 // Drop from the FRONT. The oldest messages matter least - usually.
62 while (size > MAX_CHARS && history.length > 2) {
63 const dropped = history.shift();
64 size -= dropped.content.length;
65 console.log('Forgot an old message to stay under the limit.');
66 }
67}

69// Week 4: ask the AI a real question.
70// A chat request is a LIST of messages. Each one says who said it.
71async function askAI() {
72 trimHistory(); // week 8: never send more than the server will read
73 const res = await fetch(AI_BASE + '/v1/chat/completions', {
74 method: 'POST', // POST = I am sending you something
75 headers: {
76 'Content-Type': 'application/json', // what I am sending is JSON
77 'Authorization': 'Bearer ' + API_KEY
78 },
79 body: JSON.stringify({
80 model: modelBox.value, // week 9: chosen on the page
81 temperature: Number(tempBox.value), // 0 = careful, 1.5 = wild
82 messages: [
83 // Week 6: the standing instruction, read before every single reply.
84 { role: 'system', content: personaBox.value },
85 // Week 7: everything said so far, oldest first.
86 ...history
87 ]
88 })
89 });
90 
91 // Week 5: fetch does NOT throw when the server says no. A 429 arrives
92 // looking just like a success until you actually check.
93 if (!res.ok) {
94 const problem = await res.json();
95 throw new Error(explain(res.status, problem));
96 }
97 
98 const data = await res.json();
99 return data.choices[0].message.content;
100}
